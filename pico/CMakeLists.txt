cmake_minimum_required(VERSION 3.20)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(pico_panel C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(pico_panel main.cpp)
option(PICO_PANEL_SOLAR "Autonomous Pico W solar receiver and 64x32 display" OFF)
option(PICO_PANEL_CHAIN_TEST "Static four-panel text test with Solar OTA recovery" OFF)
option(PICO_PANEL_CHAIN_ARROW "Four-panel one-shot moving arrow test" OFF)
if(PICO_PANEL_CHAIN_ARROW)
    if(NOT PICO_PANEL_CHAIN_TEST)
        message(FATAL_ERROR "Arrow test requires chain test")
    endif()
    target_compile_definitions(pico_panel PRIVATE PICO_PANEL_CHAIN_ARROW=1)
endif()
if(PICO_PANEL_CHAIN_TEST)
    if(NOT PICO_PANEL_SOLAR)
        message(FATAL_ERROR "Chain test requires Solar networking/OTA recovery")
    endif()
    target_compile_definitions(pico_panel PRIVATE PICO_PANEL_CHAIN_TEST=1)
endif()
if(PICO_PANEL_SOLAR)
    if(NOT PICO_BOARD STREQUAL "pico_w")
        message(FATAL_ERROR "Solar target requires pico_w")
    endif()
    target_compile_definitions(pico_panel PRIVATE PICO_PANEL_SOLAR=1 PICO_STDIO_USB_STDOUT_TIMEOUT_US=0 PICO_STACK_SIZE=0x1000)
    target_include_directories(pico_panel PRIVATE ${CMAKE_CURRENT_LIST_DIR})
    target_link_libraries(pico_panel pico_cyw43_arch_lwip_poll pico_unique_id hardware_flash hardware_watchdog pico_mbedtls)
    set(PANEL_VERSION "0.4.2-solar-readings" CACHE STRING "Solar firmware version")
    option(PANEL_OTA_TEST_CRASH "Bench only: watchdog crash on a trial OTA boot" OFF)
    if(PANEL_OTA_TEST_CRASH)
        target_compile_definitions(pico_panel PRIVATE PANEL_OTA_TEST_CRASH=1)
    endif()
    target_compile_definitions(pico_panel PRIVATE PANEL_VERSION="${PANEL_VERSION}")
    target_link_options(pico_panel PRIVATE "-L${CMAKE_CURRENT_LIST_DIR}/boot/linker")
    pico_set_linker_script(pico_panel ${CMAKE_CURRENT_LIST_DIR}/boot/linker/app2040.ld)
    add_executable(panel_bootloader boot/main.cpp)
    target_include_directories(panel_bootloader PRIVATE ${CMAKE_CURRENT_LIST_DIR})
    target_link_libraries(panel_bootloader pico_stdlib hardware_flash hardware_watchdog hardware_resets cmsis_core)
    target_compile_definitions(panel_bootloader PRIVATE PICO_STACK_SIZE=0x1000)
    target_link_options(panel_bootloader PRIVATE "-L${CMAKE_CURRENT_LIST_DIR}/boot/linker")
    pico_set_linker_script(panel_bootloader ${CMAKE_CURRENT_LIST_DIR}/boot/linker/boot2040.ld)
    pico_enable_stdio_usb(panel_bootloader 0)
    pico_enable_stdio_uart(panel_bootloader 0)
    pico_add_extra_outputs(panel_bootloader)
endif()
option(PICO_PANEL_REFRESH_NO_CONFIG "Experimental refresh without repeated config writes" OFF)
option(PICO_PANEL_REFRESH_CONTINUOUS "Experimental continuously scanned LAT-prefix refresh" OFF)
option(PICO_PANEL_REFRESH_NO_GAP "Remove only gaps from the no-config refresh" OFF)
if(PICO_PANEL_REFRESH_NO_GAP AND (NOT PICO_PANEL_REFRESH_NO_CONFIG OR PICO_PANEL_REFRESH_CONTINUOUS))
    message(FATAL_ERROR "No-gap requires no-config mode and excludes continuous mode")
endif()
if(PICO_PANEL_REFRESH_NO_CONFIG AND PICO_PANEL_REFRESH_CONTINUOUS)
    message(FATAL_ERROR "Choose only one experimental refresh mode")
endif()
target_compile_definitions(pico_panel PRIVATE PICO_PANEL_REFRESH_NO_CONFIG=$<BOOL:${PICO_PANEL_REFRESH_NO_CONFIG}>)
target_compile_definitions(pico_panel PRIVATE PICO_PANEL_REFRESH_CONTINUOUS=$<BOOL:${PICO_PANEL_REFRESH_CONTINUOUS}>)
target_compile_definitions(pico_panel PRIVATE PICO_PANEL_REFRESH_NO_GAP=$<BOOL:${PICO_PANEL_REFRESH_NO_GAP}>)
target_include_directories(pico_panel PRIVATE ${CMAKE_CURRENT_LIST_DIR}/generated)
target_compile_definitions(pico_panel PRIVATE CFG_TUD_CDC_RX_BUFSIZE=1024
    CFG_TUD_CDC_TX_BUFSIZE=256 PICO_STDIO_USB_TASK_INTERVAL_US=100)
pico_generate_pio_header(pico_panel ${CMAKE_CURRENT_LIST_DIR}/panel_wave.pio)
target_link_libraries(pico_panel pico_stdlib hardware_pio hardware_dma hardware_clocks)
pico_enable_stdio_usb(pico_panel 1)
pico_enable_stdio_uart(pico_panel 0)
pico_set_program_name(pico_panel "ICND1065L Pico USB video")
if(PICO_PANEL_SOLAR)
    pico_set_program_version(pico_panel "${PANEL_VERSION}")
elseif(PICO_PANEL_REFRESH_CONTINUOUS)
    pico_set_program_version(pico_panel "0.2.3-test-scan-prefix")
elseif(PICO_PANEL_REFRESH_NO_GAP)
    pico_set_program_version(pico_panel "0.2.4-test-no-gap")
elseif(PICO_PANEL_REFRESH_NO_CONFIG)
    pico_set_program_version(pico_panel "0.2.1-test-no-config")
else()
    pico_set_program_version(pico_panel "0.2.0")
endif()
pico_add_extra_outputs(pico_panel)
